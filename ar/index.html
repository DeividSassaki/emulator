const target = document.getElementById("targetZelda"); const pivot = document.getElementById("pivotEntity"); const objeto = document.getElementById("objectEntity"); const modelo = document.getElementById("heroModelObject"); const audio = document.getElementById("arAudio"); const fullscreenButton = document.getElementById("fullscreenButton"); const parametros = new URLSearchParams(window.location.search); const pasta = parametros.get("pasta") || "zelda"; const CONFIG_URL = "./marcadores/${pasta}/config.json"; const VELOCIDADE_ROTACAO = 0.01; let config = null; let marcadorVisivel = false; let arrastando = false; let ultimoX = 0; // ============================================================ // CARREGAR CONFIG // ============================================================ async function carregarConfig() { try { const resposta = await fetch(`${CONFIG_URL}?${Date.now()}`); if (!resposta.ok) { throw new Error("Erro ao carregar config.json"); } config = await resposta.json(); aplicarConfiguracao(); } catch (erro) { console.error( "Erro ao carregar configuração:", erro ); } } // ============================================================ // APLICAR CONFIGURAÇÃO // ============================================================ function aplicarConfiguracao() { if (!config) { return; } // ======================================================== // PIVOT // ======================================================== pivot.object3D.position.set( Number(config.pivot?.x ?? 0), Number(config.pivot?.y ?? 0), Number(config.pivot?.z ?? 0) ); pivot.object3D.rotation.set( THREE.MathUtils.degToRad( Number(config.pivotRotation?.x ?? 0) ), THREE.MathUtils.degToRad( Number(config.pivotRotation?.y ?? 0) ), THREE.MathUtils.degToRad( Number(config.pivotRotation?.z ?? 0) ) ); // ======================================================== // OBJETO // // IMPORTANTE: // Exatamente igual à calibração. // NÃO colocamos o objeto dentro do pivot. // ======================================================== objeto.object3D.position.set( Number(config.position?.x ?? 0), Number(config.position?.y ?? 0), Number(config.position?.z ?? 0) ); objeto.object3D.rotation.set( THREE.MathUtils.degToRad( Number(config.rotation?.x ?? 0) ), THREE.MathUtils.degToRad( Number(config.rotation?.y ?? 0) ), THREE.MathUtils.degToRad( Number(config.rotation?.z ?? 0) ) ); // ======================================================== // ESCALA // ======================================================== const escala = Number(config.scale ?? 1); modelo.object3D.scale.set( escala, escala, escala ); // ======================================================== // ÁUDIO // ======================================================== if (config.audio) { audio.src = `./marcadores/${pasta}/${config.audio}`; audio.load(); } } // ============================================================ // MARCADOR ENCONTRADO // ============================================================ target.addEventListener( "targetFound", () => { marcadorVisivel = true; console.log("Marcador encontrado"); tocarAudio(); } ); // ============================================================ // MARCADOR PERDIDO // ============================================================ target.addEventListener( "targetLost", () => { marcadorVisivel = false; console.log("Marcador perdido"); } ); // ============================================================ // ÁUDIO // ============================================================ function tocarAudio() { if (!audio.src) { return; } audio.currentTime = 0; audio.play().catch( erro => { console.log( "Áudio aguardando interação:", erro ); } ); } // ============================================================ // PEGAR O CANVAS DO A-FRAME // ============================================================ function configurarToque() { const canvas = document.querySelector( ".a-canvas" ); if (!canvas) { setTimeout( configurarToque, 500 ); return; } // ======================================================== // TOUCH START // ======================================================== canvas.addEventListener( "touchstart", evento => { if ( evento.touches.length !== 1 ) { return; } arrastando = true; ultimoX = evento.touches[0].clientX; // Primeiro toque libera o áudio if (marcadorVisivel) { tocarAudio(); } }, { passive: true } ); // ======================================================== // TOUCH MOVE // ======================================================== canvas.addEventListener( "touchmove", evento => { if ( !arrastando || evento.touches.length !== 1 ) { return; } const atualX = evento.touches[0].clientX; const deltaX = atualX - ultimoX; ultimoX = atualX; girarObjeto( deltaX ); }, { passive: true } ); // ======================================================== // TOUCH END // ======================================================== canvas.addEventListener( "touchend", () => { arrastando = false; }, { passive: true } ); canvas.addEventListener( "touchcancel", () => { arrastando = false; }, { passive: true } ); } // ============================================================ // GIRAR OBJETO AO REDOR DO PIVOT // ============================================================ function girarObjeto(deltaX) { const quantidade = deltaX * VELOCIDADE_ROTACAO; // -------------------------------------------------------- // Posição atual do objeto em relação ao pivot // -------------------------------------------------------- const posicaoObjeto = objeto.object3D.position.clone(); const posicaoPivot = pivot.object3D.position.clone(); const distancia = posicaoObjeto .clone() .sub(posicaoPivot); // -------------------------------------------------------- // Rotação em torno do eixo X do pivot // -------------------------------------------------------- const rotacao = new THREE.Quaternion(); rotacao.setFromAxisAngle( new THREE.Vector3(1, 0, 0), -quantidade ); // -------------------------------------------------------- // Nova posição // -------------------------------------------------------- distancia.applyQuaternion( rotacao ); objeto.object3D.position.copy( posicaoPivot .clone() .add(distancia) ); // -------------------------------------------------------- // Nova orientação // -------------------------------------------------------- objeto.object3D.quaternion.premultiply( rotacao ); } // ============================================================ // FULLSCREEN // ============================================================ fullscreenButton.addEventListener( "click", async () => { try { if (!document.fullscreenElement) { await document.documentElement.requestFullscreen(); } else { await document.exitFullscreen(); } } catch (erro) { console.error( "Erro no fullscreen:", erro ); } } ); // ============================================================ // INICIAR // ============================================================ carregarConfig(); configurarToque();
